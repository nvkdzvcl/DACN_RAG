"""Grounded RAG answering: rewrite -> retrieve -> select sentences -> verify -> recheck sources.

The model never writes answer text. It only picks source/sentence IDs; the backend copies those
sentences verbatim. Decisions such as "ask the customer for a missing condition" or "sources
conflict" are made by the model from the retrieved sources, not by per-question regex rules.
Regex remains only for input safety (instruction filtering) and structural source checks.
"""
import json
import os
import re
import unicodedata
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, ValidationError

from app.db.session import SessionLocal
from app.models.support import KnowledgeDocument
from app.rag.ollama import chat
from app.rag.vector_store import document_lock, vector_store

CLARIFICATION = "Bạn đang hỏi về vấn đề nào và trong trường hợp cụ thể nào? Vui lòng nêu rõ để tôi tra đúng chính sách."
CONFLICT = "Tài liệu có quy định mâu thuẫn cho cùng trường hợp. Tôi chưa thể xác nhận quy định áp dụng; bạn vui lòng yêu cầu nhân viên kiểm tra."
INSTRUCTION = re.compile(
    r"(?i)(?:<\s*/?\s*(?:system|developer|assistant)\b|"
    r"&lt;\s*/?\s*(?:system|developer|assistant)\b|"
    r"\[/?INST\]|<\|(?:im_start|im_end|start_header_id|end_header_id)\|>|"
    r"system\s+(?:update|override)|dữ liệu không đáng tin|"
    r"\b(?:bỏ qua|phớt lờ|ignore|disregard)\b[^.!?\n]{0,160}"
    r"\b(?:chỉ dẫn|hướng dẫn|quy tắc|hệ thống|câu hỏi|instructions?|rules?|system|prompts?|questions?)\b|"
    r"\b(?:trợ lý|chatbot|assistant|AI)\b[^.!?\n]{0,100}\b(?:phải|hãy|must|should)\b|"
    r"bỏ nguồn|(?:luôn|mọi)\s+(?:trả lời|đáp án)|chế độ quản trị)"
)
ROLE_BLOCK = re.compile(
    r"(?is)(?:<\s*(system|developer|assistant)\b[^>]*>.*?(?:<\s*/\s*\1\s*>|$)|"
    r"&lt;\s*(system|developer|assistant)\b.*?&gt;.*?(?:&lt;\s*/\s*\2\s*&gt;|$)|"
    r"\[INST\].*?(?:\[/INST\]|$)|<\|im_start\|>.*?(?:<\|im_end\|>|$)|"
    r"<\|start_header_id\|>.*?(?:<\|eot_id\|>|$))"
)
# A customer asking for a phone/account number must receive one printed next to its label, never generic advice.
CONTACT_NUMBERS = [
    (re.compile(r"số điện thoại|hotline|phone number", re.IGNORECASE),
     re.compile(r"(?:điện thoại|hotline|phone)[^.!?\n\d]{0,60}\d(?:[ .()-]?\d){5,}", re.IGNORECASE)),
    (re.compile(r"số tài khoản|account number", re.IGNORECASE),
     re.compile(r"(?:tài khoản|account)[^.!?\n\d]{0,60}\d(?:[ -]?\d){5,}", re.IGNORECASE)),
]
MAX_SOURCE_CHARS = 1200


class Evidence(BaseModel):
    model_config = ConfigDict(extra="forbid")
    source_id: int = Field(ge=1)
    quote: str = Field(min_length=1, max_length=1500)


class SentenceChoice(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)
    source_id: int = Field(ge=1)
    sentence_id: int = Field(ge=1)


class StandaloneQuestion(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)
    question: str = Field(min_length=1, max_length=400)


class SourceSelection(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)
    reason: str = Field(default="", max_length=500)
    decision: Literal["answer", "clarify", "conflict", "insufficient"] = "answer"
    # ponytail: three excerpts fit the 4000-character message limit; add packing for longer answers.
    citations: list[SentenceChoice] = Field(max_length=3)


class GroundedAnswer(BaseModel):
    model_config = ConfigDict(extra="forbid")
    supported: bool
    answer: str = Field(max_length=4000)
    citations: list[Evidence] = Field(max_length=5)


class AnswerReview(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)
    reason: str = Field(min_length=1, max_length=1500)
    question_resolved: bool
    sources_consistent: bool
    claims_supported: bool
    needs_clarification: bool
    repair_citations: list[SentenceChoice] = Field(max_length=3)

def nfc(text):
    return unicodedata.normalize("NFC", text)


def source_sentences(text):
    parts = re.split(r"(?<=[.!?])\s+", text)
    scoped = re.match(r"^(?:\*\*)?[^.!?:\n]{1,80}:(?:\*\*)?\s", text) is not None
    # ponytail: explicit paragraph scopes and anaphora only; broader coreference needs independently evaluated parsing.
    return [" ".join(parts[:index + 1]) if scoped or re.match(
        r"(?i)^(trường hợp (này|đó)|khi đó|sau đó|điều này|đây là|this (case|condition)|in (this|that) case)\b", part) or re.search(
        r"(?i)\b(không đủ|chưa đủ|not (?:enough|sufficient))\b", part)
        else part for index, part in enumerate(parts)]


def unresolved_notices(text):
    # ponytail: structural check for two numbered notices with different values and no stated precedence.
    text = nfc(text).casefold()
    notices = set(re.findall(r"(?:thông báo|chính sách|văn bản)\s+([a-z0-9]+)\b", text))
    if len(notices) < 2 or not re.search(
            r"không (?:ghi|có|nêu|xác định)[^.!?]{0,100}(?:hiệu lực|phiên bản|ưu tiên|thay thế)", text):
        return False
    values = {}
    for number, unit in re.findall(r"\b(\d[\d.,]*)\s*(đồng|tháng|ngày|giờ)\b", text):
        values.setdefault(unit, set()).add(number)
    return any(len(numbers) > 1 for numbers in values.values())


def opposing_policy(evidence, texts):
    # ponytail: structural check for the same scoped clause appearing both permitted and negated.
    def clauses(text):
        for sentence in re.split(r"(?<=[.!?])\s+", nfc(text).casefold()):
            sentence = re.sub(r"\[\d+\]\s*", "", sentence).strip().rstrip(".!?")
            if ":" not in sentence or re.search(r"thay thế|ưu tiên|ngoại lệ|trước ngày|từ ngày", sentence):
                continue
            positive = re.sub(r"\bkhông\s+(?=được\b|cho phép\b|bao gồm\b|áp dụng\b)", "", sentence)
            yield " ".join(positive.split()), positive != sentence
    selected = {clause for item in evidence for clause in clauses(item.quote)}
    return any((key, not negative) in selected for text in texts for key, negative in clauses(text))


def clean_history(history):
    return [{"role": m["role"], "content": m["content"][:300]} for m in (history or [])[-4:]
            if not INSTRUCTION.search(nfc(m["content"]))]


def standalone_question(query, history):
    """Resolve a follow-up into a self-contained question; one model call only when the customer has history."""
    user_turns = [m["content"] for m in history if m["role"] == "user"]
    if not user_turns:
        return query
    fallback = "\n".join([*(turn[:240] for turn in user_turns), query])
    response = chat([{"role": "system", "content": REWRITE_PROMPT}, {"role": "user", "content": json.dumps(
        {"HISTORY": history, "CÂU HỎI CUỐI": query}, ensure_ascii=False)}], StandaloneQuestion.model_json_schema())
    try:
        rewritten = " ".join(StandaloneQuestion.model_validate_json(response).question.split())
    except ValidationError:
        return fallback
    # A rewrite carrying instructions is discarded; the raw turns are still safe retrieval input.
    return rewritten if rewritten and not INSTRUCTION.search(nfc(rewritten)) else fallback


def build_candidates(useful):
    """Split each retrieved chunk into selectable sentences, dropping instruction-like or role-tagged text.

    sentence_id=1 is the longest clean prefix of the chunk; ids >= 2 are individual sentences.
    """
    sources = [{"source_id": i + 1, "text": r["content"][:MAX_SOURCE_CHARS]} for i, r in enumerate(useful)]
    sentences = [[source["text"], *re.split(r"(?<=[.!?])\s+", source["text"])] for source in sources]
    choices = []
    for source_id, parts in enumerate(sentences, 1):
        blocked = [match.span() for match in ROLE_BLOCK.finditer(parts[0])]
        unsafe = set()
        offset = 0
        for sentence_id, part in enumerate(parts[1:], 2):
            start = parts[0].find(part, offset)
            offset = start + len(part)
            if INSTRUCTION.search(nfc(part)) or any(start < end and offset > begin for begin, end in blocked):
                unsafe.add(sentence_id)
        clean_prefix = []
        for sentence_id, part in enumerate(parts[1:], 2):
            if sentence_id in unsafe:
                break
            clean_prefix.append(part)
        parts[0] = " ".join(clean_prefix)
        choices.append({"source_id": source_id, "sentences": [{"sentence_id": sentence_id, "text": text}
                        for sentence_id, text in enumerate(parts, 1) if text and sentence_id not in unsafe]})
    for source, choice in zip(sources, choices):
        source["text"] = " ".join(f"[{part['sentence_id']}] {part['text']}" for part in choice["sentences"] if part["sentence_id"] != 1)
    return sources, choices, sentences


def sentence_choice_schema(choices):
    # Bind sentence IDs to their source in Ollama's grammar; provider output is still validated afterwards.
    return {"oneOf": [
        {"type": "object", "additionalProperties": False, "required": ["source_id", "sentence_id"],
         "properties": {"source_id": {"type": "integer", "enum": [choice["source_id"]]},
                        "sentence_id": {"type": "integer", "enum": [part["sentence_id"] for part in choice["sentences"]]}}}
        for choice in choices if choice["sentences"]]}


def select_evidence(question, history, choices, choice_schema):
    schema = SourceSelection.model_json_schema()
    schema["required"] = ["reason", "decision", "citations"]
    schema["$defs"]["SentenceChoice"] = choice_schema
    response = chat([{"role": "system", "content": SELECTION_PROMPT}, {"role": "user", "content": json.dumps(
        {"HISTORY": history, "SOURCES": choices, "QUESTION": question}, ensure_ascii=False)}], schema)
    try:
        return SourceSelection.model_validate_json(response)
    except ValidationError:
        return None


def assemble_evidence(selected, useful, sentences, choices):
    """Copy selected sentences verbatim; returns (evidence, citation metadata) or None on any invalid ID."""
    evidence, citations, seen = [], [], set()
    for citation in selected:
        if citation.source_id > len(useful):
            return None
        if citation.sentence_id not in {part["sentence_id"] for part in choices[citation.source_id - 1]["sentences"]}:
            return None
        identity = (citation.source_id, citation.sentence_id)
        if identity in seen:
            continue
        seen.add(identity)
        source = useful[citation.source_id - 1]
        parts = sentences[citation.source_id - 1]
        quote = " ".join((parts[0] if citation.sentence_id == 1 else
                          source_sentences(source["content"][:MAX_SOURCE_CHARS])[citation.sentence_id - 2]).split())
        if not quote or INSTRUCTION.search(nfc(quote)):
            return None
        if any(item.source_id == citation.source_id and quote in item.quote for item in evidence):
            continue
        # A wider excerpt from the same source replaces the narrower ones it contains.
        retained = [(item, metadata) for item, metadata in zip(evidence, citations)
                    if not (item.source_id == citation.source_id and item.quote in quote)]
        evidence = [item for item, _ in retained]
        citations = [metadata for _, metadata in retained]
        evidence.append(Evidence(source_id=citation.source_id, quote=quote))
        citations.append({k: source[k] for k in ("chunk_id", "document_id", "index_version", "source", "score", "page", "location")}
                         | {"quote": quote})
    return evidence, citations


def review_answer(question, history, sources, answer, choice_schema):
    # ponytail: same-model review reduces errors, not a proof; add independent human evaluation before acceptance.
    schema = AnswerReview.model_json_schema()
    schema["$defs"]["SentenceChoice"] = choice_schema
    payload = {"QUESTION": question, "HISTORY": history, "CANDIDATE": answer.model_dump(), "SOURCES": sources}
    response = chat([{"role": "system", "content": REVIEW_PROMPT}, {"role": "user", "content": json.dumps(
        payload, ensure_ascii=False)}], schema)
    try:
        return AnswerReview.model_validate_json(response)
    except ValidationError:
        return None


def sources_current(db, sources):
    for source in sources:
        document = db.get(KnowledgeDocument, source.get("document_id"), populate_existing=True)
        if not document or document.status != "indexed" or document.index_version != source.get("index_version"):
            return False
    return True


def answer_question(query, top_k=5, history=None, db=None):
    if db is None:
        with SessionLocal() as session:
            return answer_question(query, top_k, history, session)
    history = clean_history(history)
    # History resolves follow-up subjects; it is never evidence for policy facts.
    question = standalone_question(query, history)
    results = vector_store.search(question, top_k, db)
    useful = [r for r in results if r["score"] >= float(os.getenv("RAG_MIN_SCORE", "0.35"))][:5]
    fallback = {"answer": ABSTENTION, "grounded": False, "citations": [], "results": results}
    clarify = fallback | {"answer": CLARIFICATION, "needs_clarification": True}
    conflict = fallback | {"answer": CONFLICT}
    required_number = next((labelled for request, labelled in CONTACT_NUMBERS if request.search(question)), None)
    if not useful or (required_number and not any(required_number.search(r["content"]) for r in useful)):
        return fallback
    sources, choices, sentences = build_candidates(useful)
    if not any(choice["sentences"] for choice in choices):
        return fallback
    choice_schema = sentence_choice_schema(choices)
    selection = select_evidence(question, history, choices, choice_schema)
    if selection is None:
        return fallback
    if selection.decision == "clarify":
        return clarify
    if selection.decision == "conflict":
        return conflict
    allowed = {(choice["source_id"], part["sentence_id"]) for choice in choices for part in choice["sentences"]}
    selected = selection.citations if selection.decision == "answer" else []
    for attempt in range(2):
        if not selected or any((item.source_id, item.sentence_id) not in allowed for item in selected):
            return fallback
        assembled = assemble_evidence(selected, useful, sentences, choices)
        if assembled is None or not assembled[0]:
            return fallback
        evidence, citations = assembled
        answer = GroundedAnswer(supported=True, answer="\n\n".join(item.quote for item in evidence), citations=evidence)
        if required_number and not required_number.search(answer.answer):
            return fallback
        if any(unresolved_notices(item.quote) for item in evidence) or opposing_policy(evidence, [s["text"] for s in sources]):
            return conflict
        review = review_answer(question, history, sources, answer, choice_schema)
        if review is None:
            return fallback
        if not review.sources_consistent:
            return fallback if review.needs_clarification else conflict
        if review.needs_clarification and not review.question_resolved:
            return clarify
        if review.question_resolved and review.claims_supported and not review.needs_clarification:
            break
        # One repair: the reviewer may point at better sentences, which are then reviewed again.
        if attempt:
            return fallback
        selected = review.repair_citations
    reviewed_sources = [{k: source[k] for k in ("document_id", "index_version")} for source in useful]
    with document_lock:
        valid = sources_current(db, reviewed_sources)
        db.commit()
    if not valid:
        return fallback | {"results": []}
    return {"answer": answer.answer.strip(), "grounded": True, "citations": citations, "results": results,
            "reviewed_sources": reviewed_sources}
